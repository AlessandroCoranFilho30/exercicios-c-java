# Exercício 1 ✅
Uma empresa possui 4 filiais contendo três produtos em todas as filiais. A Empresa quer saber:
O Produto de maior volume e qual filial pertence
A Menor quantidade de produto
Total de itens de todos os produtos

## Java
```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int produtos = 3;
        int[][] estoque = new int[4][3];
        int filialMaior = 0, filialMenor = 0;
        int produtoMaior = 0, produtoMenor = 0;
        int soma = 0;

        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < produtos; j++) {
                System.out.printf("Quantidade produto Filial %d (Produto %d): ", i + 1, j + 1);
                estoque[i][j] = scanner.nextInt();
            }
        }

        int maiorVol = estoque[0][0];
        int menorVol = estoque[0][0];

        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < produtos; j++) {
                int qtd = estoque[i][j];

                soma += qtd;

                if (qtd > maiorVol) {
                    maiorVol = qtd;
                    filialMaior = i;
                    produtoMaior = j;
                }
                if (qtd < menorVol) {
                    menorVol = qtd;
                    filialMenor = i;
                    produtoMenor = j;
                }
            }
        }

        System.out.println("----------------------------------");
        System.out.printf("Maior volume: Produto %d, Filial %d (%d unidades)%n", produtoMaior + 1, filialMaior + 1, maiorVol);
        System.out.printf("Menor quantidade: Produto %d, Filial %d (%d unidade(s))%n", produtoMenor + 1, filialMenor + 1, menorVol);
        System.out.println("Total de itens de todos os produtos: " + soma + " unidades");

        scanner.close();
    }
}
```

## Java
```java
import java.util.Scanner;

public class ExerciciosMatrizVetor {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int[] v1 = new int[5];
        int[] v2 = new int[10];
        int[][] matriz = new int[4][3];
        int[][] matrizResultante = new int[4][3];

        // 1. Leitura do primeiro vetor (5 elementos)
        System.out.println("--- Digitação do Vetor 1 (5 elementos) ---");
        for (int i = 0; i < 5; i++) {
            System.out.print("Vetor 1 [" + i + "]: ");
            v1[i] = scanner.nextInt();
        }

        // 2. Leitura do segundo vetor (10 elementos)
        System.out.println("\n--- Digitação do Vetor 2 (10 elementos) ---");
        for (int i = 0; i < 10; i++) {
            System.out.print("Vetor 2 [" + i + "]: ");
            v2[i] = scanner.nextInt();
        }

        // 3. Leitura da Matriz 4x3
        System.out.println("\n--- Digitação da Matriz 4x3 ---");
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 3; j++) {
                System.out.print("Matriz [" + i + "][" + j + "]: ");
                matriz[i][j] = scanner.nextInt();
            }
        }

        // 4. Encontrar o MAIOR elemento do primeiro vetor
        int maiorV1 = v1[0];
        for (int i = 1; i < 5; i++) {
            if (v1[i] > maiorV1) {
                maiorV1 = v1[i];
            }
        }

        // 5. Encontrar o MENOR elemento do segundo vetor
        int menorV2 = v2[0];
        for (int i = 1; i < 10; i++) {
            if (v2[i] < menorV2) {
                menorV2 = v2[i];
            }
        }

        // Fator a ser somado a cada elemento da matriz
        int fator = maiorV1 * menorV2;
        System.out.println("\nMaior do Vetor 1 (" + maiorV1 + ") * Menor do Vetor 2 (" + menorV2 + ") = " + fator);

        // 6. Gerar a matriz resultante
        System.out.println("\n--- Matriz Resultante (4x3) ---");
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 3; j++) {
                matrizResultante[i][j] = matriz[i][j] + fator;
                System.out.print(matrizResultante[i][j] + "\t");
            }
            System.out.println();
        }

        // 7. Soma dos elementos PARES de cada LINHA da matriz resultante
        System.out.println("\n--- Soma dos elementos PARES de cada LINHA ---");
        for (int i = 0; i < 4; i++) {
            int somaParesLinha = 0;
            for (int j = 0; j < 3; j++) {
                if (matrizResultante[i][j] % 2 == 0) {
                    somaParesLinha += matrizResultante[i][j];
                }
            }
            System.out.println("Linha " + i + ": Soma dos pares = " + somaParesLinha);
        }

        // 8. Quantidade de elementos entre 10 e 20 em cada COLUNA da matriz resultante
        System.out.println("\n--- Quantidade de elementos entre 10 e 20 em cada COLUNA ---");
        for (int j = 0; j < 3; j++) {
            int qtdIntervalo = 0;
            for (int i = 0; i < 4; i++) {
                if (matrizResultante[i][j] >= 10 && matrizResultante[i][j] <= 20) {
                    qtdIntervalo++;
                }
            }
            System.out.println("Coluna " + j + ": " + qtdIntervalo + " elemento(s) entre 10 e 20");
        }

        scanner.close();
    }
}
```

## C
```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int v1[5];
    int v2[10];
    int matriz[4][3];
    int matriz_resultante[4][3];
    int i, j;

    printf("--- Digitacao do Vetor 1 (5 elementos) ---\n");
    for (i = 0; i < 5; i++) {
        printf("Vetor 1 [%d]: ", i);
        scanf("%d", &v1[i]);
    }

    printf("\n--- Digitacao do Vetor 2 (10 elementos) ---\n");
    for (i = 0; i < 10; i++) {
        printf("Vetor 2 [%d]: ", i);
        scanf("%d", &v2[i]);
    }

    printf("\n--- Digitacao da Matriz 4x3 ---\n");
    for (i = 0; i < 4; i++) {
        for (j = 0; j < 3; j++) {
            printf("Matriz [%d][%d]: ", i, j);
            scanf("%d", &matriz[i][j]);
        }
    }

    int maior_v1 = v1[0];
    for (i = 1; i < 5; i++) {
        if (v1[i] > maior_v1) {
            maior_v1 = v1[i];
        }
    }

    int menor_v2 = v2[0];
    for (i = 1; i < 10; i++) {
        if (v2[i] < menor_v2) {
            menor_v2 = v2[i];
        }
    }

    int fator = maior_v1 * menor_v2;
    printf("\nMaior do Vetor 1 (%d) * Menor do Vetor 2 (%d) = %d\n", maior_v1, menor_v2, fator);

    printf("\n--- Matriz Resultante (4x3) ---\n");
    for (i = 0; i < 4; i++) {
        for (j = 0; j < 3; j++) {
            matriz_resultante[i][j] = matriz[i][j] + fator;
            printf("%d\t", matriz_resultante[i][j]);
        }
        printf("\n");
    }

    printf("\n--- Soma dos elementos PARES de cada LINHA ---\n");
    for (i = 0; i < 4; i++) {
        int soma_pares_linha = 0;
        for (j = 0; j < 3; j++) {
            if (matriz_resultante[i][j] % 2 == 0) {
                soma_pares_linha += matriz_resultante[i][j];
            }
        }
        printf("Linha %d: Soma dos pares = %d\n", i, soma_pares_linha);
    }

    printf("\n--- Quantidade de elementos entre 10 e 20 em cada COLUNA ---\n");
    for (j = 0; j < 3; j++) {
        int qtd_intervalo = 0;
        for (i = 0; i < 4; i++) {
            if (matriz_resultante[i][j] >= 10 && matriz_resultante[i][j] <= 20) {
                qtd_intervalo++;
            }
        }
        printf("Coluna %d: %d elemento(s) entre 10 e 20\n", j, qtd_intervalo);
    }

    return 0;
}
```
